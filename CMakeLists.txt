cmake_minimum_required(VERSION 3.31)
option(PM_BUILD_GRAPHICS_EXPERIMENTS "Build the Windows M01 feasibility experiments" OFF)
option(PM_BUILD_BGFX_EXPERIMENT "Build the pinned bgfx D3D11 import experiment" OFF)
option(PM_BUILD_SPOUT_EXPERIMENT "Build the M01 independent-process Spout experiment" OFF)
option(PM_BUILD_DECODER_EXPERIMENT "Build the M01 D3D11 hardware decoder/import experiment" OFF)
set(VCPKG_MANIFEST_FEATURES "")
if(PM_BUILD_BGFX_EXPERIMENT)
  if(NOT PM_BUILD_GRAPHICS_EXPERIMENTS)
    message(FATAL_ERROR "PM_BUILD_BGFX_EXPERIMENT requires PM_BUILD_GRAPHICS_EXPERIMENTS")
  endif()
  list(APPEND VCPKG_MANIFEST_FEATURES graphics-experiments)
endif()
if(PM_BUILD_SPOUT_EXPERIMENT)
  if(NOT PM_BUILD_GRAPHICS_EXPERIMENTS OR NOT PM_BUILD_BGFX_EXPERIMENT)
    message(FATAL_ERROR "PM_BUILD_SPOUT_EXPERIMENT requires graphics and bgfx experiments")
  endif()
  list(APPEND VCPKG_MANIFEST_FEATURES spout-experiments)
endif()
if(PM_BUILD_DECODER_EXPERIMENT)
  if(NOT PM_BUILD_GRAPHICS_EXPERIMENTS OR NOT PM_BUILD_BGFX_EXPERIMENT)
    message(FATAL_ERROR "PM_BUILD_DECODER_EXPERIMENT requires graphics and bgfx experiments")
  endif()
  list(APPEND VCPKG_MANIFEST_FEATURES decoder-experiments)
endif()
project(MapGyver VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(PM_BUILD_PLATFORM "Build native GPU/window adapters" OFF)
option(PM_BUILD_MEDIA "Build media adapters" OFF)
option(PM_BUILD_UI "Build the static web editor" OFF)
option(PM_BUILD_HARDWARE_TESTS "Build tests requiring a qualified hardware profile" OFF)
option(PM_BUILD_NDI "Build the owner-approved optional NDI adapter" OFF)
include(CTest)
get_cmake_property(cache_variables CACHE_VARIABLES)
foreach(variable IN LISTS cache_variables)
  if(variable MATCHES "^PM_BUILD_" AND NOT variable MATCHES "^PM_BUILD_(PLATFORM|MEDIA|UI|HARDWARE_TESTS|NDI|GRAPHICS_EXPERIMENTS|BGFX_EXPERIMENT|SPOUT_EXPERIMENT|DECODER_EXPERIMENT)$")
    message(FATAL_ERROR "Unknown MapGyver feature selection: ${variable}")
  endif()
endforeach()
foreach(feature PLATFORM MEDIA UI HARDWARE_TESTS NDI)
  if(PM_BUILD_${feature})
    message(FATAL_ERROR "PM_BUILD_${feature} is not implemented/qualified yet; it cannot be silently enabled.")
  endif()
endforeach()
if(MSVC AND MSVC_VERSION LESS 1944)
  message(FATAL_ERROR "MapGyver requires MSVC 19.44 or newer.")
endif()
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
find_package(nlohmann_json 3.12.0 CONFIG REQUIRED)
find_package(Boost 1.92 REQUIRED COMPONENTS beast)
set(MAPGYVER_VERSION "${PROJECT_VERSION}-dev")
set(metadata_dir "${CMAKE_CURRENT_BINARY_DIR}/generated/mapgyver")
file(MAKE_DIRECTORY "${metadata_dir}")
configure_file(cmake/build_config.hpp.in "${metadata_dir}/build_config.hpp" @ONLY)
add_custom_target(mapgyver_source_metadata
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/build_metadata.py"
    --root "${CMAKE_CURRENT_SOURCE_DIR}" --output "${metadata_dir}/source_revision.hpp"
  BYPRODUCTS "${metadata_dir}/source_revision.hpp"
  COMMENT "Capture actual source identity" VERBATIM)
add_library(mapgyver_core STATIC libs/core/src/build_info.cpp)
add_dependencies(mapgyver_core mapgyver_source_metadata)
target_include_directories(mapgyver_core PUBLIC libs/core/include PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(mapgyver_core PRIVATE MAPGYVER_BUILD_CONFIGURATION="$<CONFIG>")
add_subdirectory(libs/domain)
add_subdirectory(libs/control)
add_subdirectory(libs/protocol)
add_library(mapgyver_network STATIC adapters/osc/network.cpp)
target_include_directories(mapgyver_network PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(mapgyver_network PUBLIC mapgyver_protocol PRIVATE Boost::beast)
if(WIN32)
  target_compile_definitions(mapgyver_network PRIVATE _WIN32_WINNT=0x0A00)
endif()
if(MSVC)
  target_compile_options(mapgyver_network PRIVATE /bigobj)
endif()
add_executable(mapgyver_engine apps/engine/main.cpp)
set_target_properties(mapgyver_engine PROPERTIES OUTPUT_NAME mapgyver-engine)
target_link_libraries(mapgyver_engine PRIVATE mapgyver_core mapgyver_protocol mapgyver_network nlohmann_json::nlohmann_json)
if(WIN32)
  target_sources(mapgyver_engine PRIVATE adapters/windows/pipe_service.cpp)
  target_include_directories(mapgyver_engine PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(mapgyver_engine PRIVATE advapi32)
endif()
foreach(owned_target mapgyver_core mapgyver_engine mapgyver_network)
  if(MSVC)
    target_compile_options(${owned_target} PRIVATE /W4 /WX /permissive- /utf-8)
  else()
    target_compile_options(${owned_target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
install(TARGETS mapgyver_engine RUNTIME DESTINATION bin)
if(PM_BUILD_GRAPHICS_EXPERIMENTS)
  if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "M01 graphics experiments require native Windows x64; portable core remains separate.")
  endif()
  add_subdirectory(experiments/graphics)
endif()
if(BUILD_TESTING)
  add_executable(mapgyver_network_fixture tests/protocol/network_fixture.cpp)
  target_link_libraries(mapgyver_network_fixture PRIVATE mapgyver_network mapgyver_core)
  if(MSVC)
    target_compile_options(mapgyver_network_fixture PRIVATE /W4 /WX /permissive- /utf-8)
  else()
    target_compile_options(mapgyver_network_fixture PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME protocol.network
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/protocol/test_network.py"
      --fixture "$<TARGET_FILE:mapgyver_network_fixture>"
      --report "${CMAKE_CURRENT_SOURCE_DIR}/out/reports/network-$<CONFIG>.json")
  set_tests_properties(protocol.network PROPERTIES TIMEOUT 120)
  if(WIN32)
    add_test(NAME protocol.engine-network
      COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/protocol/test_network.py"
        --engine "$<TARGET_FILE:mapgyver_engine>"
        --report "${CMAKE_CURRENT_SOURCE_DIR}/out/reports/engine-network-$<CONFIG>.json")
    set_tests_properties(protocol.engine-network PROPERTIES TIMEOUT 120)
  endif()
  add_test(NAME diagnostics.contract
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_diagnostics.py"
      --engine "$<TARGET_FILE:mapgyver_engine>" --source "${CMAKE_CURRENT_SOURCE_DIR}"
      --configuration "$<CONFIG>")
  add_test(NAME build.source-identity
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_build_metadata.py"
      --script "${CMAKE_CURRENT_SOURCE_DIR}/tools/build_metadata.py")
  add_test(NAME build.feature-rejection
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_configuration.py"
      --source "${CMAKE_CURRENT_SOURCE_DIR}" --cmake "${CMAKE_COMMAND}" --generator "${CMAKE_GENERATOR}")
  add_test(NAME planning.contract
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/validate_plan.py")
  add_test(NAME ci.quality-gate
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_quality_gate.py"
      --script "${CMAKE_CURRENT_SOURCE_DIR}/tools/ci/quality_gate.py")
  set_tests_properties(ci.quality-gate PROPERTIES TIMEOUT 30)
  set_tests_properties(diagnostics.contract build.source-identity planning.contract PROPERTIES TIMEOUT 30)
  set_tests_properties(build.feature-rejection PROPERTIES TIMEOUT 150)
  if(WIN32)
    add_test(NAME protocol.windows-pipe
      COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/protocol/test_engine_pipe.py"
        --engine "$<TARGET_FILE:mapgyver_engine>"
        --report "${CMAKE_CURRENT_SOURCE_DIR}/out/reports/pipe-$<CONFIG>.json")
    set_tests_properties(protocol.windows-pipe PROPERTIES TIMEOUT 120)
  endif()
endif()
