add_library(mapgyver_runtime STATIC src/outputs.cpp)
target_include_directories(mapgyver_runtime PUBLIC include)
target_link_libraries(mapgyver_runtime PUBLIC mapgyver_domain)
if(MSVC)
  target_compile_options(mapgyver_runtime PRIVATE /W4 /WX /permissive- /utf-8)
else()
  target_compile_options(mapgyver_runtime PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING)
  add_executable(mapgyver_output_binding_test ../../tests/render/test_outputs.cpp)
  target_link_libraries(mapgyver_output_binding_test PRIVATE mapgyver_runtime mapgyver_control)
  if(MSVC)
    target_compile_options(mapgyver_output_binding_test PRIVATE /W4 /WX /permissive- /utf-8)
  else()
    target_compile_options(mapgyver_output_binding_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME runtime.outputs COMMAND mapgyver_output_binding_test)
  set_tests_properties(runtime.outputs PROPERTIES TIMEOUT 30)
endif()
