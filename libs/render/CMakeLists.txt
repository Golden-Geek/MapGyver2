add_library(mapgyver_render STATIC src/plan.cpp)
target_include_directories(mapgyver_render PUBLIC include)
target_link_libraries(mapgyver_render PUBLIC mapgyver_control)
if(MSVC)
  target_compile_options(mapgyver_render PRIVATE /W4 /WX /permissive- /utf-8)
else()
  target_compile_options(mapgyver_render PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING)
  add_executable(mapgyver_render_plan_test ../../tests/render/test_plan.cpp)
  target_link_libraries(mapgyver_render_plan_test PRIVATE mapgyver_render)
  if(MSVC)
    target_compile_options(mapgyver_render_plan_test PRIVATE /W4 /WX /permissive- /utf-8)
  else()
    target_compile_options(mapgyver_render_plan_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME render.plan COMMAND mapgyver_render_plan_test)
  set_tests_properties(render.plan PROPERTIES TIMEOUT 30)
endif()
