add_library(mapgyver_protocol STATIC src/service.cpp src/osc.cpp src/oscquery.cpp)
target_include_directories(mapgyver_protocol PUBLIC include)
target_link_libraries(mapgyver_protocol PUBLIC mapgyver_control)
if(MSVC)
  target_compile_options(mapgyver_protocol PRIVATE /W4 /WX /permissive- /utf-8)
else()
  target_compile_options(mapgyver_protocol PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING)
  add_executable(mapgyver_protocol_service_test ../../tests/protocol/test_service.cpp)
  target_link_libraries(mapgyver_protocol_service_test PRIVATE mapgyver_protocol)
  if(MSVC)
    target_compile_options(mapgyver_protocol_service_test PRIVATE /W4 /WX /permissive- /utf-8)
  else()
    target_compile_options(mapgyver_protocol_service_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME protocol.service COMMAND mapgyver_protocol_service_test)
  set_tests_properties(protocol.service PROPERTIES TIMEOUT 30)
  foreach(contract osc oscquery)
    add_executable(mapgyver_protocol_${contract}_test ../../tests/protocol/test_${contract}.cpp)
    target_link_libraries(mapgyver_protocol_${contract}_test PRIVATE mapgyver_protocol)
    if(MSVC)
      target_compile_options(mapgyver_protocol_${contract}_test PRIVATE /W4 /WX /permissive- /utf-8)
    else()
      target_compile_options(mapgyver_protocol_${contract}_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME protocol.${contract} COMMAND mapgyver_protocol_${contract}_test)
    set_tests_properties(protocol.${contract} PROPERTIES TIMEOUT 30)
  endforeach()
endif()
