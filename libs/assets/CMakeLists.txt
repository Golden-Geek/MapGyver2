add_library(mapgyver_assets STATIC src/import.cpp)
target_include_directories(mapgyver_assets PUBLIC include)
target_link_libraries(mapgyver_assets PUBLIC mapgyver_control)
if(MSVC)
  target_compile_options(mapgyver_assets PRIVATE /W4 /WX /permissive- /utf-8)
else()
  target_compile_options(mapgyver_assets PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING)
  add_executable(mapgyver_assets_import_test ../../tests/media/test_import.cpp)
  target_link_libraries(mapgyver_assets_import_test PRIVATE mapgyver_assets)
  if(MSVC)
    target_compile_options(mapgyver_assets_import_test PRIVATE /W4 /WX /permissive- /utf-8)
  else()
    target_compile_options(mapgyver_assets_import_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME assets.image_import COMMAND mapgyver_assets_import_test)
  set_tests_properties(assets.image_import PROPERTIES TIMEOUT 30)
endif()
