add_library(mapgyver_control STATIC src/project.cpp src/registry.cpp src/wire.cpp)
target_include_directories(mapgyver_control PUBLIC include)
target_link_libraries(mapgyver_control PUBLIC mapgyver_domain nlohmann_json::nlohmann_json)
if(MSVC)
  target_compile_options(mapgyver_control PRIVATE /W4 /WX /permissive- /utf-8)

endif()
