add_library(mapgyver_control STATIC src/project.cpp)
target_include_directories(mapgyver_control PUBLIC include)
target_link_libraries(mapgyver_control PUBLIC mapgyver_domain)
if(MSVC)
  target_compile_options(mapgyver_control PRIVATE /W4 /WX /permissive- /utf-8)
else()
  target_compile_options(mapgyver_control PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING)
  add_executable(mapgyver_control_project_test ../../tests/domain/test_project.cpp)
  target_link_libraries(mapgyver_control_project_test PRIVATE mapgyver_control)
  if(MSVC)
    target_compile_options(mapgyver_control_project_test PRIVATE /W4 /WX /permissive- /utf-8)
  else()
    target_compile_options(mapgyver_control_project_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME control.project COMMAND mapgyver_control_project_test)
  set_tests_properties(control.project PROPERTIES TIMEOUT 30)
endif()
